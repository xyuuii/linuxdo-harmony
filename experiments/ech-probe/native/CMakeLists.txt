cmake_minimum_required(VERSION 3.16)
project(echprobe LANGUAGES C CXX)
if(NOT OHOS_ARCH STREQUAL "arm64-v8a")
  message(FATAL_ERROR "This isolated ECH prototype supports arm64-v8a only")
endif()
if(NOT EXISTS "${ECH_RUST_LIB}")
  message(FATAL_ERROR "Build the Rust static library first and set ECH_RUST_LIB")
endif()
add_library(echprobe SHARED ech_probe.cpp)
target_compile_features(echprobe PRIVATE cxx_std_17)
target_link_libraries(echprobe PRIVATE "${ECH_RUST_LIB}" libace_napi.z.so dl m)
